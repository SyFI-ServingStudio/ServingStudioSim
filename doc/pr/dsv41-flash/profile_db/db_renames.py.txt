"""Apply the post-merge kind/backend/value renames to a profile.db copy.

Phase B input: a COPY of tmp/dsv41/merge/profile_v41_backup.db (schema v2).
Run this BEFORE `kernel-profile migrate-db`: v2 tables key rows by their args
columns, so renaming a table, a backend or an args value is plain SQL. (On a v3
DB an args-value rename would also have to recompute `args_hash`; this script
refuses v3 tables for value renames.)

    unset VIRTUAL_ENV
    cp tmp/dsv41/merge/profile_v41_backup.db tmp/dsv41/merge/work.db
    uv run python tmp/dsv41/merge/db_renames.py tmp/dsv41/merge/work.db [--dry-run]
    uv run python -m launcher kernel-profile migrate-db tmp/dsv41/merge/work.db

The mapping has three sources (see tmp/dsv41/merge/renames.md):
- main 50be2ba / 4cae722: V4 kinds renamed by mechanism, two V4 kinds merged
  into DSA kinds, model-named backends renamed. main's v3 DB already holds
  these rows under the new names; renaming them here makes the backup line up
  with main so merge-db skips them instead of re-creating the old tables.
- main 2e85d96 / 592bfd6: *_vllm_fork backends folded into canonical names.
  Only gemm_fp32_output/torch_cublas_vllm_fork occurs in the backup.
- this branch (Phase A step 2): the two V4.1 kinds, the wo_a batched_gemm
  backend, and the MXFP4 weight_format DType wire value.

Every step prints its row counts. Idempotent: a step whose source is absent
is reported and skipped.
"""

from __future__ import annotations

import argparse
import sqlite3
import sys
from dataclasses import dataclass

# ── table (kind) renames: old table -> new table ─────────────────────────────
TABLE_RENAMES: dict[str, str] = {
    # main 50be2ba (V4 kinds named by mechanism)
    "deepseek_v4_fused_inv_rope_fp8_quant": "inv_rope_fp8_quant",
    "deepseek_v4_fused_q_kv_rmsnorm": "q_kv_rms_norm",
    "deepseek_v4_indexer_mqa_logits_prefill": "dsa_compressed_mqa_logits_prefill",
    "deepseek_v4_indexer_topk_prefill": "dsa_compressed_topk_prefill",
    "deepseek_v4_indexer_q_rope_quant": "dsa_indexer_q_rope_quant_weight_fold",
    "deepseek_v4_packed_cache_gather": "packed_kv_cache_gather",
    "deepseek_v4_qnorm_rope_kv_insert": "qnorm_rope_kv_insert",
    "deepseek_v4_sparse_attn_compress_store": "kv_compress_store",
    "deepseek_v4_sparse_mla_decode": "compressed_sparse_mla_decode",
    "deepseek_v4_sparse_mla_prefill": "compressed_sparse_mla_prefill",
    "deepseek_v4_terminal_mhc_head": "mhc_terminal_head",
    # this branch (V4.1 kinds named by mechanism)
    "deepseek_v41_mega_attn": "compressed_sparse_mla_rope_cast",
    "deepseek_v41_qnorm_rope_kv_insert": "q_pad_kv_rope_mxfp8_insert",
}


# ── table merges: rows of a source table move into an existing kind ──────────
@dataclass(frozen=True)
class TableMerge:
    source: str
    target: str
    backend_map: dict[str, str]


TABLE_MERGES: tuple[TableMerge, ...] = (
    # main 50be2ba: same callable and args, so the V4 rows are a layout of the kind.
    TableMerge(
        "deepseek_v4_indexer_mqa_logits_decode",
        "dsa_paged_mqa_logits_decode",
        {"vllm_deepgemm_fp8": "deepgemm_fp8"},
    ),
    TableMerge(
        "deepseek_v4_indexer_topk_decode",
        "dsa_persistent_topk_decode",
        {"vllm_cuda": "vllm_fork_cuda"},
    ),
)


# ── backend renames inside a (post-rename) table ─────────────────────────────
@dataclass(frozen=True)
class BackendRename:
    table: str
    old: str
    new: str
    # On a key collision with an existing `new` row: "keep_existing" leaves the
    # old-named row in place (main 592bfd6's rule); "replace" deletes the
    # existing row so the renamed one wins.
    on_conflict: str = "keep_existing"


BACKEND_RENAMES: tuple[BackendRename, ...] = (
    # main 50be2ba
    BackendRename("kv_compress_store", "vllm_deepseek_v4_cutedsl", "vllm_cutedsl"),
    BackendRename("kv_compress_store", "vllm_deepseek_v4_triton", "vllm_triton"),
    BackendRename("packed_kv_cache_gather", "vllm_deepseek_v4_cutedsl", "vllm_cutedsl"),
    BackendRename("batched_gemm", "torch_mla_q_absorb_glm52", "torch_mla_q_absorb"),
    BackendRename("batched_gemm", "torch_mla_v_up_glm52", "torch_mla_v_up"),
    # main 2e85d96 fold. The k=5120 bf16 shapes exist under both names in the
    # backup (both measured by this branch; main has no k=5120 rows). The V4.1
    # arch read the fork rows, which reproduce the captured NvJet/splitKreduce
    # kernels, so they replace the container rows to keep its timings.
    BackendRename("gemm_fp32_output", "torch_cublas_vllm_fork", "torch_cublas", "replace"),
    # this branch
    BackendRename(
        "batched_gemm",
        "deepgemm_mxfp8_einsum_dsv41_wo_a",
        "deepgemm_mxfp8_einsum_grouped_o_proj",
    ),
)


# ── args-value renames (v2 only) ─────────────────────────────────────────────
@dataclass(frozen=True)
class ValueRename:
    table: str
    backend: str
    column: str
    old: str
    new: str


VALUE_RENAMES: tuple[ValueRename, ...] = (
    # nvfp4_fused_moe.weight_format is a DType on main; MXFP4 is DType.MXFP4_E2M1.
    ValueRename(
        "nvfp4_fused_moe", "flashinfer_trtllm_sm100_mxfp4", "weight_format", "mxfp4_ue8m0", "mxfp4_e2m1"
    ),
)

_NON_KEY = {
    "id",
    "backend",
    "profiler_git_hash",
    "profiler_run_at",
    "cuda_version",
    "driver_version",
    "backend_version",
    "verified",
    "time_ms",
    "tflops",
    "memory_bandwidth_gbps",
    "algbw_gbps",
    "busbw_gbps",
    "energy_j",
    "is_outlier",
    "retry_count",
    "outlier_reason",
    "created_at",
    "args_hash",
    "run_key",
}


def _q(name: str) -> str:
    return '"' + name.replace('"', '""') + '"'


def _tables(conn: sqlite3.Connection) -> set[str]:
    return {r[0] for r in conn.execute("SELECT name FROM sqlite_master WHERE type='table'")}


def _cols(conn: sqlite3.Connection, table: str) -> list[str]:
    return [r[1] for r in conn.execute(f"PRAGMA table_info({_q(table)})")]


def _key_cols(conn: sqlite3.Connection, table: str) -> list[str]:
    """gpu_name + args columns (v2: the UNIQUE key minus backend)."""
    return [c for c in _cols(conn, table) if c not in _NON_KEY]


def _count(conn: sqlite3.Connection, table: str, backend: str | None = None) -> int:
    if backend is None:
        return conn.execute(f"SELECT COUNT(*) FROM {_q(table)}").fetchone()[0]
    return conn.execute(
        f"SELECT COUNT(*) FROM {_q(table)} WHERE backend = ?", (backend,)
    ).fetchone()[0]


def _match(alias_a: str, alias_b: str, cols: list[str]) -> str:
    return " AND ".join(f"{alias_a}.{_q(c)} IS {alias_b}.{_q(c)}" for c in cols)


def apply(conn: sqlite3.Connection) -> None:
    tables = _tables(conn)
    for old, new in TABLE_RENAMES.items():
        if old not in tables:
            print(f"table  {old}: absent, skip")
            continue
        if new in tables:
            sys.exit(f"table  {old} -> {new}: target already exists; resolve by hand")
        n = _count(conn, old)
        conn.execute(f"ALTER TABLE {_q(old)} RENAME TO {_q(new)}")
        tables = _tables(conn)
        print(f"table  {old} -> {new}: {n} rows")

    for merge in TABLE_MERGES:
        if merge.source not in tables:
            print(f"merge  {merge.source}: absent, skip")
            continue
        if merge.target not in tables:
            sys.exit(f"merge  {merge.source} -> {merge.target}: target table missing")
        src_cols = [c for c in _cols(conn, merge.source) if c != "id"]
        if set(src_cols) != {c for c in _cols(conn, merge.target) if c != "id"}:
            sys.exit(f"merge  {merge.source} -> {merge.target}: column sets differ")
        keys = _key_cols(conn, merge.target)
        moved = kept = 0
        for old_backend, new_backend in merge.backend_map.items():
            select = ", ".join("?" if c == "backend" else f"s.{_q(c)}" for c in src_cols)
            cur = conn.execute(
                f"INSERT INTO {_q(merge.target)} ({', '.join(_q(c) for c in src_cols)}) "
                f"SELECT {select} FROM {_q(merge.source)} s WHERE s.backend = ? AND NOT EXISTS "
                f"(SELECT 1 FROM {_q(merge.target)} t WHERE t.backend = ? AND {_match('t', 's', keys)})",
                (new_backend, old_backend, new_backend),
            )
            moved += cur.rowcount
        total = _count(conn, merge.source)
        kept = total - moved
        if kept:
            sys.exit(f"merge  {merge.source}: {kept} rows collide in {merge.target}; resolve by hand")
        conn.execute(f"DROP TABLE {_q(merge.source)}")
        tables = _tables(conn)
        print(f"merge  {merge.source} -> {merge.target}: {moved} rows")

    for rn in BACKEND_RENAMES:
        if rn.table not in tables or _count(conn, rn.table, rn.old) == 0:
            print(f"backend {rn.table}/{rn.old}: absent, skip")
            continue
        keys = _key_cols(conn, rn.table)
        collide = f"EXISTS (SELECT 1 FROM {_q(rn.table)} t WHERE t.backend = ? AND {_match('t', 'o', keys)})"
        n_collide = conn.execute(
            f"SELECT COUNT(*) FROM {_q(rn.table)} o WHERE o.backend = ? AND {collide}",
            (rn.old, rn.new),
        ).fetchone()[0]
        if n_collide and rn.on_conflict == "replace":
            conn.execute(
                f"DELETE FROM {_q(rn.table)} WHERE backend = ? AND id IN ("
                f"SELECT t.id FROM {_q(rn.table)} t JOIN {_q(rn.table)} o "
                f"ON o.backend = ? AND {_match('t', 'o', keys)})",
                (rn.new, rn.old),
            )
        cur = conn.execute(
            f"UPDATE {_q(rn.table)} AS o SET backend = ? WHERE o.backend = ? AND NOT {collide}",
            (rn.new, rn.old, rn.new),
        )
        left = _count(conn, rn.table, rn.old)
        print(
            f"backend {rn.table}/{rn.old} -> {rn.new}: {cur.rowcount} renamed, "
            f"{n_collide} collided ({rn.on_conflict}), {left} left under the old name"
        )

    for vr in VALUE_RENAMES:
        if vr.table not in tables:
            print(f"value  {vr.table}: absent, skip")
            continue
        if "args_hash" in _cols(conn, vr.table):
            sys.exit(f"value  {vr.table}: v3 table; apply value renames before migrate-db")
        cur = conn.execute(
            f"UPDATE {_q(vr.table)} SET {_q(vr.column)} = ? WHERE backend = ? AND {_q(vr.column)} = ?",
            (vr.new, vr.backend, vr.old),
        )
        print(f"value  {vr.table}/{vr.backend}.{vr.column} {vr.old} -> {vr.new}: {cur.rowcount} rows")


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    parser.add_argument("db")
    parser.add_argument("--dry-run", action="store_true", help="report, then roll back")
    args = parser.parse_args()
    conn = sqlite3.connect(args.db)
    conn.execute("BEGIN IMMEDIATE")
    try:
        apply(conn)
    except BaseException:
        conn.rollback()
        raise
    if args.dry_run:
        conn.rollback()
        print("dry run: rolled back")
    else:
        conn.commit()
    conn.close()


if __name__ == "__main__":
    main()
